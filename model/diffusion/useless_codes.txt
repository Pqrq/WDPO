    def q_sample(self, x_start, t, noise=None):
        """""" 
        Diffuse x_start to timestep t using the forward q-process:
          x_t = sqrt_alphas_cumprod[t] * x_start + sqrt(1 - alphas_cumprod[t]) * noise
        If noise is None, sample standard normal noise and return (x_t, noise).
        x_start: [B, Ta, Da] or [B, D_flat] shaped as same as actions
        t: tensor of shape [B] (int timesteps)
        """""" 
        if noise is None:
            noise = torch.randn_like(x_start)
        # broadcast extracts to x_start shape
        sqrt_alpha_cum = extract(self.sqrt_alphas_cumprod, t, x_start.shape)
        sqrt_one_minus_alpha = extract(self.sqrt_one_minus_alphas_cumprod, t, x_start.shape)
        x_t = sqrt_alpha_cum * x_start + sqrt_one_minus_alpha * noise
        return x_t, noise

    def predict_eps(self, x_t, t, cond):
        """""" 
        Policy mean in noise-space: ε̂θ(x_t, t, cond)
        Calls the actor network — assumes actor signature actor(x, t, cond)
        Returns tensor shaped like x_t: [B, Ta, Da]
        """""" 
        return self.actor(x_t, t, cond=cond)

    def eps_to_x0(self, x_t, eps, t):
        """""" 
        Deterministic reconstruction of x0 (action) from (x_t, eps):
          x0 = (x_t - sqrt(1-α̅_t) * eps) / sqrt(α̅_t)
        """""" 
        sqrt_alpha_cum = extract(self.sqrt_alphas_cumprod, t, x_t.shape)
        sqrt_one_minus_alpha = extract(self.sqrt_one_minus_alphas_cumprod, t, x_t.shape)
        x0 = (x_t - sqrt_one_minus_alpha * eps) / (sqrt_alpha_cum + 1e-12)
        return x0
